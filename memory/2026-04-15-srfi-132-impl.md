```scheme
;; predicates.scm -- sorted predicates for lists and vectors

(define (list-sorted? < lis)
  "Return #t if LIS is sorted according to the ordering predicate <.
An empty list and a single-element list are always sorted.

Examples:
  (list-sorted? < '(1 2 3))    => #t
  (list-sorted? < '(1 3 2))    => #f
  (list-sorted? < '())         => #t

Parameters:
  < : procedure
  lis : list
Returns: boolean
Category: srfi-132
Keywords: sorted, ordered, predicate, check"
  (or (null? lis)
      (null? (cdr lis))
      (let loop ((prev (car lis)) (rest (cdr lis)))
        (or (null? rest)
            (let ((cur (car rest)))
              (and (not (< cur prev))
                   (loop cur (cdr rest))))))))

(define vector-sorted?
  (case-lambda
    "Return #t if vector V (or the subrange [START, END)) is sorted
according to the ordering predicate <.

Examples:
  (vector-sorted? < '#(1 2 3))       => #t
  (vector-sorted? < '#(1 3 2))       => #f
  (vector-sorted? < '#(5 1 2 3 9) 1 4)  => #t

Parameters:
  < : procedure
  v : vector
  start : integer (optional, default 0)
  end : integer (optional, default (vector-length v))
Returns: boolean
Category: srfi-132
Keywords: sorted, ordered, predicate, check, vector"
    ((< v) (%vector-sorted? < v 0 (vector-length v)))
    ((< v start) (%vector-sorted? < v start (vector-length v)))
    ((< v start end) (%vector-sorted? < v start end))))

(define (%vector-sorted? < v start end)
  (or (<= (- end start) 1)
      (let loop ((i (+ start 1)))
        (or (>= i end)
            (and (not (< (vector-ref v i) (vector-ref v (- i 1))))
```

```scheme
;; list-merge.scm -- stable merge of two sorted lists

(define (list-merge < lis1 lis2)
  "Merge two sorted lists into a single sorted list, allocating
fresh cons cells. Stable: elements from LIS1 come before equal
elements from LIS2.

Examples:
  (list-merge < '(1 3 5) '(2 4 6))   => (1 2 3 4 5 6)
  (list-merge < '(1 3) '(2 3 4))     => (1 2 3 3 4)
  (list-merge < '() '(1 2))          => (1 2)

Parameters:
  < : procedure
  lis1 : list
  lis2 : list
Returns: list
Category: srfi-132
Keywords: merge, sorted, combine, stable"
  (cond
    ((null? lis1) lis2)
    ((null? lis2) lis1)
    ((< (car lis2) (car lis1))
     (cons (car lis2) (list-merge < lis1 (cdr lis2))))
    (else
     (cons (car lis1) (list-merge < (cdr lis1) lis2)))))

(define (list-merge! < lis1 lis2)
  "Merge two sorted lists by relinking existing cons cells.
The input lists should not be used after this call. Stable:
elements from LIS1 come before equal elements from LIS2.

Examples:
  (list-merge! < (list 1 3 5) (list 2 4 6))  => (1 2 3 4 5 6)

Parameters:
  < : procedure
  lis1 : list
  lis2 : list
Returns: list
Category: srfi-132
Keywords: merge, sorted, combine, stable, destructive, linear update"
  (cond
    ((null? lis1) lis2)
    ((null? lis2) lis1)
    ((< (car lis2) (car lis1))
     (set-cdr! lis2 (list-merge! < lis1 (cdr lis2)))
     lis2)
    (else
     (set-cdr! lis1 (list-merge! < (cdr lis1) lis2))
     lis1)))
```

### Task 3: List sort

**Files:**
- Create: `stdlib/lib/srfi/132/list-sort.scm`

**Step 1: Implement list sort**

Create `stdlib/lib/srfi/132/list-sort.scm`. This is the most complex piece — bottom-up natural merge sort.

The algorithm:
1. `%collect-runs` scans the list for ascending and descending runs, reversing descending runs in place. Returns a list of sorted run heads.
2. `%merge-pairs` pairwise merges the run list until one remains.
3. `list-sort` copies the list first (to avoid mutating input), then sorts destructively.
4. `list-sort!` sorts destructively directly.

```scheme
;; list-sort.scm -- natural merge sort for lists

;; Reverse a list segment ending at tail, destructively.
;; Returns the new head.
(define (%reverse-segment! head tail)
  (let loop ((prev tail) (cur head))
    (if (eq? cur tail)
        prev
        (let ((next (cdr cur)))
          (set-cdr! cur prev)
          (loop cur next)))))

;; Collect natural runs from a list. Destructive: reverses descending
;; runs in place. Returns a list of run-head pairs ((head . tail) ...).
;; Each run is a sorted sublist; tail is the last pair of the run.
(define (%collect-runs < lis)
  (if (or (null? lis) (null? (cdr lis)))
      (list lis)
      (let loop ((cur lis) (runs '()))
        (if (null? cur)
            (reverse runs)
            (if (null? (cdr cur))
                (reverse (cons cur runs))
                ;; Determine run direction
                (let ((next (cdr cur)))
                  (if (< (car next) (car cur))
                      ;; Descending run: scan to end, then reverse
                      (let desc ((end next))
                        (if (and (pair? (cdr end))
                                 (< (cadr end) (car end)))
                            (desc (cdr end))
                            (let ((rest (cdr end)))
                              (set-cdr! end '())
                              (let ((head (%reverse-segment! cur '())))
                                (loop rest (cons head runs))))))
                      ;; Ascending run: scan to end
                      (let asc ((end next))
                        (if (and (pair? (cdr end))
                                 (not (< (cadr end) (car end))))
                            (asc (cdr end))
                            (let ((rest (cdr end)))
                              (set-cdr! end '())
                              (loop rest (cons cur runs))))))))))))

;; Pairwise merge a list of sorted sublists until one remains.
(define (%merge-pairs < runs)
  (cond
    ((null? runs) '())
    ((null? (cdr runs)) (car runs))
    (else
     (%merge-pairs <
       (let loop ((rs runs))
         (cond
           ((null? rs) '())
           ((null? (cdr rs)) rs)
           (else
            (cons (list-merge! < (car rs) (cadr rs))
                  (loop (cddr rs))))))))))

(define (list-sort! < lis)
  "Sort LIS by the ordering predicate <. May reuse the cons cells
of the input list. The original list should not be used after
this call. Uses natural merge sort: O(n log n) worst case, O(n)
for already-sorted input.

Examples:
  (list-sort! < (list 3 1 4 1 5 9))  => (1 1 3 4 5 9)

Parameters:
  < : procedure
  lis : list
Returns: list
Category: srfi-132
Keywords: sort, order, destructive, linear update, merge sort"
  (if (or (null? lis) (null? (cdr lis)))
      lis
      (%merge-pairs < (%collect-runs < lis))))

(define (list-sort < lis)
  "Return a list containing the elements of LIS in sorted order
according to the ordering predicate <. The original list is not
modified. Uses natural merge sort: O(n log n) worst case.

Examples:
  (list-sort < '(3 1 4 1 5 9))  => (1 1 3 4 5 9)
  (list-sort > '(3 1 4 1 5))    => (5 4 3 1 1)
  (list-sort < '())              => ()

Parameters:
  < : procedure
  lis : list
Returns: list
Category: srfi-132
Keywords: sort, order, merge sort, stable"
  (list-sort! < (list-copy lis)))

;; Stable variants are identical — merge sort is stable.
(define list-stable-sort list-sort)
(define list-stable-sort! list-sort!)
```

```scheme
;; vector-merge.scm -- merge two sorted vectors

(define (%vector-merge < v1 s1 e1 v2 s2 e2)
  ;; Merge v1[s1..e1) and v2[s2..e2) into a new vector.
  (let* ((len1 (- e1 s1))
         (len2 (- e2 s2))
         (result (make-vector (+ len1 len2))))
    (%vector-merge-into! < result 0 v1 s1 e1 v2 s2 e2)
    result))

(define (%vector-merge-into! < target tstart v1 s1 e1 v2 s2 e2)
  ;; Merge v1[s1..e1) and v2[s2..e2) into target starting at tstart.
  (let loop ((i s1) (j s2) (k tstart))
    (cond
      ((>= i e1)
       ;; Copy remaining from v2.
       (let copy ((j j) (k k))
         (when (< j e2)
           (vector-set! target k (vector-ref v2 j))
           (copy (+ j 1) (+ k 1)))))
      ((>= j e2)
       ;; Copy remaining from v1.
       (let copy ((i i) (k k))
         (when (< i e1)
           (vector-set! target k (vector-ref v1 i))
           (copy (+ i 1) (+ k 1)))))
      ((< (vector-ref v2 j) (vector-ref v1 i))
       (vector-set! target k (vector-ref v2 j))
       (loop i (+ j 1) (+ k 1)))
      (else
       ;; v1 element <= v2 element: take from v1 (stable)
       (vector-set! target k (vector-ref v1 i))
       (loop (+ i 1) j (+ k 1))))))

(define vector-merge
  (case-lambda
    "Merge two sorted vectors (or subranges) into a new sorted vector.
Stable: elements from V1 come before equal elements from V2.

Examples:
  (vector-merge < '#(1 3 5) '#(2 4 6))  => #(1 2 3 4 5 6)

Parameters:
  < : procedure
  v1 : vector
  v2 : vector
  start1 : integer (optional)
  end1 : integer (optional)
  start2 : integer (optional)
  end2 : integer (optional)
Returns: vector
Category: srfi-132
Keywords: merge, sorted, combine, stable, vector"
    ((< v1 v2)
     (%vector-merge < v1 0 (vector-length v1) v2 0 (vector-length v2)))
    ((< v1 v2 start1)
     (%vector-merge < v1 start1 (vector-length v1) v2 0 (vector-length v2)))
    ((< v1 v2 start1 end1)
     (%vector-merge < v1 start1 end1 v2 0 (vector-length v2)))
    ((< v1 v2 start1 end1 start2)
     (%vector-merge < v1 start1 end1 v2 start2 (vector-length v2)))
    ((< v1 v2 start1 end1 start2 end2)
     (%vector-merge < v1 start1 end1 v2 start2 end2))))

(define vector-merge!
  (case-lambda
    "Merge sorted vectors FROM1 and FROM2 (or subranges) into target
vector TO starting at START. Stable: elements from FROM1 come
before equal elements from FROM2.

Parameters:
  < : procedure
  to : vector
  from1 : vector
  from2 : vector
  start : integer (optional, default 0)
  start1 : integer (optional)
  end1 : integer (optional)
  start2 : integer (optional)
  end2 : integer (optional)
Returns: unspecified
Category: srfi-132
Keywords: merge, sorted, combine, stable, vector, destructive"
    ((< to from1 from2)
     (%vector-merge-into! < to 0 from1 0 (vector-length from1)
                          from2 0 (vector-length from2)))
    ((< to from1 from2 start)
     (%vector-merge-into! < to start from1 0 (vector-length from1)
                          from2 0 (vector-length from2)))
    ((< to from1 from2 start start1)
     (%vector-merge-into! < to start from1 start1 (vector-length from1)
                          from2 0 (vector-length from2)))
    ((< to from1 from2 start start1 end1)
     (%vector-merge-into! < to start from1 start1 end1
                          from2 0 (vector-length from2)))
    ((< to from1 from2 start start1 end1 start2)
     (%vector-merge-into! < to start from1 start1 end1
                          from2 start2 (vector-length from2)))
    ((< to from1 from2 start start1 end1 start2 end2)
     (%vector-merge-into! < to start from1 start1 end1
                          from2 start2 end2))))
```

```scheme
;; vector-sort.scm -- bottom-up merge sort for vectors

(define (%vector-sort! < v start end)
  ;; Bottom-up merge sort on v[start..end) using a temp buffer.
  ;; Alternates between v and temp to avoid unnecessary copies.
  (let ((len (- end start)))
    (when (> len 1)
      (let ((temp (make-vector len)))
        ;; Copy subrange into temp for initial source
        (let init ((i 0))
          (when (< i len)
            (vector-set! temp i (vector-ref v (+ start i)))
            (init (+ i 1))))
        ;; src and dst alternate each pass.
        ;; After each pass, merged runs live in dst.
        (let pass ((width 1) (src temp) (dst v) (dst-off start) (src-off 0))
          (if (>= width len)
              ;; Done. If result is in temp, copy back to v.
              (when (eq? src v)
                ;; dst=temp has the result, copy to v
                (let cp ((i 0))
                  (when (< i len)
                    (vector-set! v (+ start i) (vector-ref temp i))
                    (cp (+ i 1)))))
              (begin
                ;; Merge pairs of width-sized runs from src into dst.
                (let merge-runs ((pos 0))
                  (when (< pos len)
                    (let* ((mid (min (+ pos width) len))
                           (right-end (min (+ pos width width) len)))
                      (%vector-merge-into! < dst (+ dst-off pos)
                                           src (+ src-off pos) (+ src-off mid)
                                           src (+ src-off mid) (+ src-off right-end))
                      (merge-runs right-end))))
                (pass (* width 2) dst src src-off dst-off))))))))

(define vector-sort!
  (case-lambda
    "Sort vector V (or the subrange [START, END)) in place using
the ordering predicate <. Uses bottom-up merge sort: O(n log n)
time, O(n) auxiliary space.

Examples:
  (let ((v (vector 3 1 4 1 5)))
    (vector-sort! < v)
    v)  => #(1 1 3 4 5)

Parameters:
  < : procedure
  v : vector
  start : integer (optional, default 0)
  end : integer (optional, default (vector-length v))
Returns: unspecified
Category: srfi-132
Keywords: sort, order, vector, destructive, merge sort"
    ((< v) (%vector-sort! < v 0 (vector-length v)))
    ((< v start) (%vector-sort! < v start (vector-length v)))
    ((< v start end) (%vector-sort! < v start end))))

(define vector-sort
  (case-lambda
    "Return a newly allocated sorted vector containing the elements
of V (or the subrange [START, END)) in order according to <.
The original vector is not modified.

Examples:
  (vector-sort < '#(3 1 4 1 5))          => #(1 1 3 4 5)
  (vector-sort < '#(9 3 1 4 7) 1 4)     => #(1 3 4)

Parameters:
  < : procedure
  v : vector
  start : integer (optional, default 0)
  end : integer (optional, default (vector-length v))
Returns: vector
Category: srfi-132
Keywords: sort, order, vector, merge sort, stable"
    ((< v) (%vector-sort-copy < v 0 (vector-length v)))
    ((< v start) (%vector-sort-copy < v start (vector-length v)))
    ((< v start end) (%vector-sort-copy < v start end))))

(define (%vector-sort-copy < v start end)
  (let ((result (vector-copy v start end)))
    (%vector-sort! < result 0 (- end start))
    result))
```

```scheme
;; dedup.scm -- remove consecutive duplicate elements

(define (list-delete-neighbor-dups = lis)
  "Return a list with consecutive equal elements (per =) removed,
keeping the first of each equal run. Allocates fresh cons cells.

Examples:
  (list-delete-neighbor-dups = '(1 1 2 2 3 3))  => (1 2 3)
  (list-delete-neighbor-dups = '(1 2 1 2))      => (1 2 1 2)

Parameters:
  = : procedure
  lis : list
Returns: list
Category: srfi-132
Keywords: deduplicate, unique, adjacent, neighbor"
  (if (null? lis)
      '()
      (let loop ((prev (car lis)) (rest (cdr lis)) (acc (list (car lis))))
        (cond
          ((null? rest) (reverse acc))
          ((= prev (car rest))
           (loop prev (cdr rest) acc))
          (else
           (loop (car rest) (cdr rest) (cons (car rest) acc)))))))

(define (list-delete-neighbor-dups! = lis)
  "Remove consecutive equal elements (per =) from LIS by relinking
cons cells. The original list should not be used after this call.

Examples:
  (list-delete-neighbor-dups! = (list 1 1 2 2 3))  => (1 2 3)

Parameters:
  = : procedure
  lis : list
Returns: list
Category: srfi-132
Keywords: deduplicate, unique, adjacent, neighbor, destructive"
  (if (null? lis)
      lis
      (let loop ((prev lis) (cur (cdr lis)))
        (cond
          ((null? cur) lis)
          ((= (car prev) (car cur))
           (set-cdr! prev (cdr cur))
           (loop prev (cdr cur)))
          (else
           (loop cur (cdr cur)))))))

(define vector-delete-neighbor-dups
  (case-lambda
    "Return a newly allocated vector with consecutive equal elements
(per =) removed from V (or the subrange [START, END)).

Examples:
  (vector-delete-neighbor-dups = '#(1 1 2 2 3))  => #(1 2 3)

Parameters:
  = : procedure
  v : vector
  start : integer (optional)
  end : integer (optional)
Returns: vector
Category: srfi-132
Keywords: deduplicate, unique, adjacent, neighbor, vector"
    ((= v) (%vector-delete-neighbor-dups = v 0 (vector-length v)))
    ((= v start) (%vector-delete-neighbor-dups = v start (vector-length v)))
    ((= v start end) (%vector-delete-neighbor-dups = v start end))))

(define (%vector-delete-neighbor-dups = v start end)
  (if (>= start end)
      '#()
      ;; First pass: compact into a temp copy, count unique elements.
      (let ((temp (make-vector (- end start))))
        (vector-set! temp 0 (vector-ref v start))
        (let loop ((i (+ start 1)) (j 1))
          (if (>= i end)
              (vector-copy temp 0 j)
              (let ((cur (vector-ref v i)))
                (if (= (vector-ref temp (- j 1)) cur)
                    (loop (+ i 1) j)
                    (begin
                      (vector-set! temp j cur)
                      (loop (+ i 1) (+ j 1))))))))))

(define vector-delete-neighbor-dups!
  (case-lambda
    "Remove consecutive equal elements (per =) from V (or the
subrange [START, END)) by compacting in place. Returns the new
logical end index (an integer), not a vector. Elements beyond
the returned index are unchanged.

Examples:
  (let ((v (vector 1 1 2 2 3)))
    (vector-delete-neighbor-dups! = v))  => 3  ; v is now #(1 2 3 2 3)

Parameters:
  = : procedure
  v : vector
  start : integer (optional)
  end : integer (optional)
Returns: integer
Category: srfi-132
Keywords: deduplicate, unique, adjacent, neighbor, vector, destructive"
    ((= v) (%vector-delete-neighbor-dups! = v 0 (vector-length v)))
    ((= v start) (%vector-delete-neighbor-dups! = v start (vector-length v)))
    ((= v start end) (%vector-delete-neighbor-dups! = v start end))))

(define (%vector-delete-neighbor-dups! = v start end)
  (if (>= start end)
      start
      (let loop ((i (+ start 1)) (j (+ start 1)))
        (if (>= i end)
            j
            (if (= (vector-ref v (- j 1)) (vector-ref v i))
                (loop (+ i 1) j)
                (begin
                  (vector-set! v j (vector-ref v i))
                  (loop (+ i 1) (+ j 1))))))))
```

**Step 1: Implement quickselect**

Create `stdlib/lib/srfi/132/select.scm`:

```scheme
;; select.scm -- selection and separation via quickselect

;; Swap two vector elements.
(define (%swap! v i j)
  (let ((tmp (vector-ref v i)))
    (vector-set! v i (vector-ref v j))
    (vector-set! v j tmp)))

;; Median-of-three pivot selection. Returns the index of the median.
(define (%median-of-3 < v a b c)
  (let ((va (vector-ref v a))
        (vb (vector-ref v b))
        (vc (vector-ref v c)))
    (cond
      ((< va vb)
       (if (< vb vc) b (if (< va vc) c a)))
      (else
       (if (< va vc) a (if (< vb vc) c b))))))

;; Three-way partition v[lo..hi) around pivot value.
;; Returns (lt-end eq-end) where:
;;   v[lo..lt-end) < pivot
;;   v[lt-end..eq-end) = pivot
;;   v[eq-end..hi) > pivot
(define (%partition3! < v lo hi pivot)
  (let loop ((lt lo) (i lo) (gt hi))
    (if (>= i gt)
        (values lt gt)
        (let ((vi (vector-ref v i)))
          (cond
            ((< vi pivot)
             (%swap! v lt i)
             (loop (+ lt 1) (+ i 1) gt))
            ((< pivot vi)
             (%swap! v i (- gt 1))
             (loop lt i (- gt 1)))
            (else
             (loop lt (+ i 1) gt)))))))

;; Quickselect: rearrange so that v[k] holds the k-th smallest
;; element of v[lo..hi). Elements below k are <= v[k], above are >= v[k].
(define (%quickselect! < v lo hi k)
  (if (>= (- hi lo) 2)
      (let* ((mid (quotient (+ lo hi) 2))
             (pivot-idx (%median-of-3 < v lo mid (- hi 1)))
             (pivot (vector-ref v pivot-idx)))
        (call-with-values
          (lambda () (%partition3! < v lo hi pivot))
          (lambda (lt-end eq-end)
            (cond
              ((< k lt-end)
               (%quickselect! < v lo lt-end k))
              ((>= k eq-end)
               (%quickselect! < v eq-end hi k))
              ;; else k is in the equal partition — done
              ))))))

(define vector-select!
  (case-lambda
    "Find the K-th smallest element of V (or the subrange
[START, END)), where K is zero-based. The vector is mutated
as a side effect of the selection algorithm.

Examples:
  (let ((v (vector 3 1 4 1 5)))
    (vector-select! < v 0))      => 1   ; minimum
  (let ((v (vector 3 1 4 1 5)))
    (vector-select! < v 4))      => 5   ; maximum
  (let ((v (vector 3 1 4 1 5)))
    (vector-select! < v 2))      => 3   ; median

Parameters:
  < : procedure
  v : vector
  k : integer
  start : integer (optional)
  end : integer (optional)
Returns: any
Category: srfi-132
Keywords: select, kth smallest, order statistic, quickselect"
    ((< v k)
     (%quickselect! < v 0 (vector-length v) k)
     (vector-ref v k))
    ((< v k start)
     (%quickselect! < v start (vector-length v) (+ start k))
     (vector-ref v (+ start k)))
    ((< v k start end)
     (%quickselect! < v start end (+ start k))
     (vector-ref v (+ start k)))))

(define vector-separate!
  (case-lambda
    "Rearrange V (or the subrange [START, END)) so that the K
smallest elements are in positions [START, START+K) and the
remaining elements are in [START+K, END). The two groups are
not necessarily sorted internally.

Examples:
  (let ((v (vector 3 1 4 1 5 9)))
    (vector-separate! < v 3)
    ;; First 3 elements are the 3 smallest (1 1 3), rest are (4 5 9)
    ;; but internal order within each group is unspecified.
    (vector-sorted? < (vector-copy v 0 3)))  => #t after sorting

Parameters:
  < : procedure
  v : vector
  k : integer
  start : integer (optional)
  end : integer (optional)
Returns: unspecified
Category: srfi-132
Keywords: separate, partition, kth smallest, quickselect"
    ((< v k)
     (%quickselect! < v 0 (vector-length v) k))
    ((< v k start)
     (%quickselect! < v start (vector-length v) (+ start k)))
    ((< v k start end)
     (%quickselect! < v start end (+ start k)))))
```

```scheme
;; median.scm -- median finding built on quickselect

(define (%default-mean a b)
  ;; SRFI-132 default: return the lesser element.
  ;; (The spec says if no mean is provided, return the smaller of the
  ;; two middle elements for even-length vectors.)
  a)

(define vector-find-median!
  (case-lambda
    "Find the median of vector V using ordering predicate <.
Returns KNIL for empty vectors. For even-length vectors,
applies MEAN (default: pick the lesser) to the two middle
elements. Mutates V as a side effect.

Examples:
  (vector-find-median! < (vector 3 1 4 1 5) 0)  => 3
  (vector-find-median! < (vector 3 1 4 5) 0)    => 3
  (vector-find-median! < (vector) 0)             => 0

Parameters:
  < : procedure
  v : vector
  knil : any
  mean : procedure (optional)
Returns: any
Category: srfi-132
Keywords: median, middle, order statistic, quickselect"
    ((< v knil)
     (%vector-find-median! < v knil %default-mean))
    ((< v knil mean)
     (%vector-find-median! < v knil mean))))

(define (%vector-find-median! < v knil mean)
  (let ((len (vector-length v)))
    (cond
      ((= len 0) knil)
      ((= len 1) (vector-ref v 0))
      ((odd? len)
       (vector-select! < v (quotient len 2)))
      (else
       ;; Even: need both middle elements.
       ;; Select the lower middle first — this partitions so that
       ;; everything below index k is <= v[k].
       (let* ((k (- (quotient len 2) 1))
              (a (vector-select! < v k))
              ;; The upper middle is the minimum of v[k+1 .. len).
              (b (let loop ((i (+ k 2)) (m (vector-ref v (+ k 1))))
                   (if (>= i len)
                       m
                       (let ((cur (vector-ref v i)))
                         (loop (+ i 1) (if (< cur m) cur m)))))))
         (mean a b))))))

(define vector-find-median
  (case-lambda
    "Find the median of vector V without modifying it. Copies V
first, then calls vector-find-median!.

Examples:
  (vector-find-median < '#(3 1 4 1 5) 0)  => 3

Parameters:
  < : procedure
  v : vector
  knil : any
  mean : procedure (optional)
Returns: any
Category: srfi-132
Keywords: median, middle, order statistic"
    ((< v knil)
     (vector-find-median! < (vector-copy v) knil %default-mean))
    ((< v knil mean)
     (vector-find-median! < (vector-copy v) knil mean))))
```
