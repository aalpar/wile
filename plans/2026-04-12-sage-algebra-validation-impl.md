```python
#!/usr/bin/env sage
"""
Validate Wile algebra libraries against SageMath.

Usage:
    sage tools/sage/verify_algebra.sage                    # both phases, live
    sage tools/sage/verify_algebra.sage --phase structures # phase 1 only
    sage tools/sage/verify_algebra.sage --phase rewriting  # phase 2 only
    sage tools/sage/verify_algebra.sage --snapshot         # write .scm files
    sage tools/sage/verify_algebra.sage --seed 123         # custom RNG seed
"""

from sage.all import *
import argparse
import subprocess
import sys
import os
import random
from datetime import date
from pathlib import Path

# ---------------------------------------------------------------------------
# Constants
# ---------------------------------------------------------------------------

WILE_BIN = os.environ.get("WILE", "./dist/darwin/arm64/wile")
SNAPSHOT_DIR = Path("test/wile/sage-generated")
DEFAULT_SEED = 42
SAGE_VERSION = version()

# ---------------------------------------------------------------------------
# Argument parsing
# ---------------------------------------------------------------------------

def parse_args():
    p = argparse.ArgumentParser(description="Sage oracle for Wile algebra validation")
    p.add_argument("--phase", choices=["structures", "rewriting"],
                   help="Run only one phase (default: both)")
    p.add_argument("--snapshot", action="store_true",
                   help="Write static .scm test files")
    p.add_argument("--seed", type=int, default=DEFAULT_SEED,
                   help=f"RNG seed (default: {DEFAULT_SEED})")
    p.add_argument("--verbose", "-v", action="store_true",
                   help="Print each test case as it runs")
    return p.parse_args()

# ---------------------------------------------------------------------------
# Wile subprocess
# ---------------------------------------------------------------------------

def run_wile(scheme_code):
    """Run Scheme code in Wile subprocess, return stdout string."""
    result = subprocess.run(
        [WILE_BIN, "-e", scheme_code],
        capture_output=True, text=True, timeout=30,
    )
    if result.returncode != 0:
        raise RuntimeError(f"Wile error: {result.stderr.strip()}")
    return result.stdout.strip()

# ---------------------------------------------------------------------------
# Snapshot writer
# ---------------------------------------------------------------------------

def scm_header(name, seed):
    """Return the standard header comment for generated .scm files."""
    return (
        f";;; Generated by tools/sage/verify_algebra.sage --snapshot\n"
        f";;; Oracle: SageMath {SAGE_VERSION}\n"
        f";;; Date: {date.today().isoformat()}\n"
        f";;; Seed: {seed}\n"
        f";;; DO NOT EDIT — regenerate with:\n"
        f";;;   sage tools/sage/verify_algebra.sage --snapshot\n"
    )

def write_snapshot(filename, test_name, imports, body, seed):
    """Write a complete .scm test file to SNAPSHOT_DIR."""
    path = SNAPSHOT_DIR / filename
    with open(path, "w") as f:
        f.write(scm_header(test_name, seed))
        f.write(f"\n(import (scheme base)\n        (chibi test)\n")
        for imp in imports:
            f.write(f"        {imp}\n")
        f.write(f")\n\n(test-begin \"{test_name}\")\n\n")
        f.write(body)
        f.write(f"\n(test-end)\n(test-exit)\n")
    print(f"  Wrote {path}")

# Main
# ---------------------------------------------------------------------------

def main():
```

```python
def validate_integer_ring(args):
    """Validate Wile's (integer-ring) against Sage's ZZ."""
    print("  integer-ring...", end=" ", flush=True)
    samples = list(range(-10, 11))
    cases = []
    failures = 0

    for a in samples:
        for b in samples:
            expected_plus = ZZ(a) + ZZ(b)
            expected_times = ZZ(a) * ZZ(b)
            expected_minus = ZZ(a) - ZZ(b)
            cases.append((a, b, "ring-plus", int(expected_plus)))
            cases.append((a, b, "ring-times", int(expected_times)))
            cases.append((a, b, "ring-minus", int(expected_minus)))
        expected_neg = -ZZ(a)
        cases.append((a, None, "ring-negate", int(expected_neg)))

    if not args.snapshot:
        scheme = "(import (wile algebra)) (let ((R (integer-ring))) "
        tests = []
        for a, b, op, expected in cases:
            if b is not None:
                tests.append(f"(display ({op} R {a} {b})) (display #\\newline)")
            else:
                tests.append(f"(display ({op} R {a})) (display #\\newline)")
        scheme += " ".join(tests) + ")"
        output = run_wile(scheme)
        results = output.strip().split("\n")
        for i, (a, b, op, expected) in enumerate(cases):
            actual = int(results[i])
            if actual != expected:
                failures += 1
                if b is not None:
                    print(f"\n    FAIL: ({op} R {a} {b}) = {actual}, expected {expected}")
                else:
                    print(f"\n    FAIL: ({op} R {a}) = {actual}, expected {expected}")
        if failures == 0:
            print(f"OK ({len(cases)} cases)")
        else:
            print(f"\n    {failures}/{len(cases)} FAILED")
    else:
        body = "(let ((R (integer-ring)))\n"
        for a, b, op, expected in cases:
            if b is not None:
                body += f"  (test {expected} ({op} R {a} {b}))\n"
            else:
                body += f"  (test {expected} ({op} R {a}))\n"
        body += ")\n"
        write_snapshot(
            "sage-structures-integer-ring-test.scm",
            "sage-structures-integer-ring",
            ["(wile algebra)"],
            body,
            args.seed,
        )

    return failures
```

```python
def validate_rational_field(args):
    """Validate Wile's (rational-field) against Sage's QQ."""
    print("  rational-field...", end=" ", flush=True)
    nums = list(set(QQ(n)/QQ(d) for n in range(-3, 4) for d in range(1, 4)))
    cases = []
    failures = 0

    for a in nums:
        for b in nums:
            cases.append((a, b, "field-plus", a + b))
            cases.append((a, b, "field-times", a * b))
            if b != 0:
                cases.append((a, b, "field-divide", a / b))
        cases.append((a, None, "field-negate", -a))
        if a != 0:
            cases.append((a, None, "field-reciprocal", 1/a))

    def fmt_rational(q):
        """Format a Sage rational as Wile expects."""
        if q.denominator() == 1:
            return str(int(q))
        return f"{int(q.numerator())}/{int(q.denominator())}"

    if not args.snapshot:
        scheme = "(import (wile algebra)) (let ((F (rational-field))) "
        tests = []
        for a, b, op, expected in cases:
            if b is not None:
                tests.append(
                    f"(display ({op} F {fmt_rational(a)} {fmt_rational(b)}))"
                    f" (display #\\newline)")
            else:
                tests.append(
                    f"(display ({op} F {fmt_rational(a)}))"
                    f" (display #\\newline)")
        scheme += " ".join(tests) + ")"
        output = run_wile(scheme)
        results = output.strip().split("\n")
        for i, (a, b, op, expected) in enumerate(cases):
            expected_str = fmt_rational(expected)
            if results[i] != expected_str:
                failures += 1
                b_str = fmt_rational(b) if b is not None else ""
                print(f"\n    FAIL: ({op} F {fmt_rational(a)} {b_str})"
                      f" = {results[i]}, expected {expected_str}")
        if failures == 0:
            print(f"OK ({len(cases)} cases)")
        else:
            print(f"\n    {failures}/{len(cases)} FAILED")
    else:
        body = "(let ((F (rational-field)))\n"
        for a, b, op, expected in cases:
            exp_str = fmt_rational(expected)
            if b is not None:
                body += (f"  (test {exp_str}"
                         f" ({op} F {fmt_rational(a)} {fmt_rational(b)}))\n")
            else:
                body += f"  (test {exp_str} ({op} F {fmt_rational(a)}))\n"
        body += ")\n"
        write_snapshot(
            "sage-structures-rational-field-test.scm",
            "sage-structures-rational-field",
            ["(wile algebra)"],
            body,
            args.seed,
        )

    return failures
```

```python
def validate_modular_ring(args):

    print("  modular-ring...", end=" ", flush=True)
    test_moduli = [2, 3, 5, 7, 12]
    cases = []
    failures = 0

    for n in test_moduli:
        R = Zmod(n)
        samples = list(range(n))
        for a in samples:
            for b in samples:
                cases.append((n, a, b, "ring-plus", int(R(a) + R(b))))
                cases.append((n, a, b, "ring-times", int(R(a) * R(b))))
                cases.append((n, a, b, "ring-minus", int(R(a) - R(b))))
            cases.append((n, a, None, "ring-negate", int(-R(a))))

    if not args.snapshot:
        for n in test_moduli:
            mod_cases = [(a, b, op, exp)
                         for (m, a, b, op, exp) in cases if m == n]
```

```python
def validate_powerset_lattice(args):
    """Validate Wile's (powerset-lattice) against Sage set operations."""
    print("  powerset-lattice...", end=" ", flush=True)
    universe = [1, 2, 3]
    from itertools import combinations
    subsets = []
    for r in range(len(universe) + 1):
        for s in combinations(universe, r):
            subsets.append(frozenset(s))
    cases = []
    failures = 0

    for a in subsets:
        for b in subsets:
            cases.append((a, b, "join", a | b))
            cases.append((a, b, "meet", a & b))
            cases.append((a, b, "leq?", a <= b))

    def fmt_set(s):
        if not s:
            return "'()"
        return "'" + "(" + " ".join(str(x) for x in sorted(s)) + ")"

    def parse_set(s):
        s = s.strip("()")
        if not s:
            return frozenset()
        return frozenset(int(x) for x in s.split())

    if not args.snapshot:
        scheme = "(import (wile algebra)) (let ((L (powerset-lattice '(1 2 3)))) "
        tests = []
        for a, b, op, expected in cases:
            if op == "join":
                tests.append(
                    f"(write (lattice-join L {fmt_set(a)} {fmt_set(b)}))"
                    f" (display #\\newline)")
            elif op == "meet":
                tests.append(
                    f"(write (lattice-meet L {fmt_set(a)} {fmt_set(b)}))"
                    f" (display #\\newline)")
            elif op == "leq?":
                tests.append(
                    f"(display (lattice-leq? L {fmt_set(a)} {fmt_set(b)}))"
                    f" (display #\\newline)")
        scheme += " ".join(tests) + ")"
        output = run_wile(scheme)
        results = output.strip().split("\n")
        for i, (a, b, op, expected) in enumerate(cases):
            if op == "leq?":
                expected_str = "#t" if expected else "#f"
                if results[i] != expected_str:
                    failures += 1
                    print(f"\n    FAIL: (lattice-leq? L {fmt_set(a)}"
                          f" {fmt_set(b)}) = {results[i]},"
                          f" expected {expected_str}")
            else:
                if parse_set(results[i]) != expected:
                    failures += 1
                    wile_op = ("lattice-join" if op == "join"
                               else "lattice-meet")
                    print(f"\n    FAIL: ({wile_op} L {fmt_set(a)}"
                          f" {fmt_set(b)}) = {results[i]},"
                          f" expected {sorted(expected)}")
        if failures == 0:
            print(f"OK ({len(cases)} cases)")
        else:
            print(f"\n    {failures}/{len(cases)} FAILED")
    else:
        body = "(let ((L (powerset-lattice '(1 2 3))))\n"
        for a, b, op, expected in cases:
            if op == "leq?":
                exp = "#t" if expected else "#f"
                body += (f"  (test {exp}"
                         f" (lattice-leq? L {fmt_set(a)} {fmt_set(b)}))\n")
            elif op == "join":
                body += (f"  (test {len(expected)}"
                         f" (length (lattice-join L"
                         f" {fmt_set(a)} {fmt_set(b)})))\n")
            elif op == "meet":
                body += (f"  (test {len(expected)}"
                         f" (length (lattice-meet L"
                         f" {fmt_set(a)} {fmt_set(b)})))\n")
        body += ")\n"
        write_snapshot(
            "sage-structures-powerset-lattice-test.scm",
            "sage-structures-powerset-lattice",
            ["(wile algebra)"],
            body,
            args.seed,
        )

    return failures
```

```python
# ---------------------------------------------------------------------------
# Phase 2: Term generation and evaluation
# ---------------------------------------------------------------------------

def random_term(operators, leaves, max_depth, depth=0):
    """Generate a random S-expression term.

    operators: list of (symbol, arity) pairs
    leaves: list of leaf values
    max_depth: maximum nesting depth
    """
    if depth >= max_depth or random.random() < 0.3:
        return random.choice(leaves)
    op, arity = random.choice(operators)
    args = [random_term(operators, leaves, max_depth, depth + 1)
            for _ in range(arity)]
    return (op, args)

def term_to_sexp(term):
    """Convert a term tuple to an S-expression string."""
    if not isinstance(term, tuple):
        return str(term)
    op, args = term
    return "(" + op + " " + " ".join(term_to_sexp(a) for a in args) + ")"

def compute_term_sage(term, evaluator):
    """Compute a term's value using a Sage evaluator dict.

    evaluator: dict mapping operator symbols to callables.
    Leaf values (non-tuples) are passed through '_leaf' function.
    """
    if not isinstance(term, tuple):
        return evaluator.get("_leaf", lambda x: x)(term)
    op, args = term
    computed_args = [compute_term_sage(a, evaluator) for a in args]
    return evaluator[op](*computed_args)

def tokenize_sexp(s):
    """Tokenize a parenthesized S-expression into top-level tokens."""
    s = s.strip()
    if s.startswith("(") and s.endswith(")"):
        s = s[1:-1].strip()
    tokens = []
    i = 0
    while i < len(s):
        if s[i] == '(':
            depth = 1
            j = i + 1
            while j < len(s) and depth > 0:
                if s[j] == '(':
                    depth += 1
                elif s[j] == ')':
                    depth -= 1
                j += 1
            tokens.append(s[i:j])
            i = j
        elif s[i] in ' \t\n':
            i += 1
        else:
            j = i
            while j < len(s) and s[j] not in ' \t\n()':
                j += 1
            tokens.append(s[i:j])
            i = j
    return tokens

def parse_sexp(s):
    """Parse an S-expression string into a term tuple.

    Atoms become integers (if numeric) or strings.
    Lists become (operator, [args...]) tuples.
    """
    s = s.strip()
    if not s.startswith("("):
        try:
            return int(s)
        except ValueError:
            try:
                return float(s)
            except ValueError:
                return s

    tokens = tokenize_sexp(s)
    if len(tokens) == 1:
        return parse_sexp(tokens[0])
    op = tokens[0]
    args = [parse_sexp(t) for t in tokens[1:]]
    return (op, args)
```

```python
    proto))
```

```python
def validate_rewriting_boolean(args, num_terms=500):
    """Validate boolean algebra rewriting soundness.

    Terms use symbolic variables (p, q, r). Soundness is verified by
    checking that the original and normalized terms have the same
    truth table over all 2^3 assignments.
    """
    print("  boolean-algebra rewriting...", end=" ", flush=True)
    operators = [('or', 2), ('and', 2), ('not', 1)]
    leaves = ['p', 'q', 'r']
    failures = 0
    snapshot_cases = []

    def compute_bool(term, env):
        if isinstance(term, str):
            return env[term]
        op, args = term
        if op == 'or':
            return compute_bool(args[0], env) or compute_bool(args[1], env)
        elif op == 'and':
            return compute_bool(args[0], env) and compute_bool(args[1], env)
        elif op == 'not':
            return not compute_bool(args[0], env)
        raise ValueError(f"Unknown op: {op}")

    def truth_table(term):
        table = []
        for p in [False, True]:
            for q in [False, True]:
                for r in [False, True]:
                    table.append(
                        compute_bool(term, {'p': p, 'q': q, 'r': r}))
        return tuple(table)

    terms = [random_term(operators, leaves, 4) for _ in range(num_terms)]

    # Batch normalize
    scheme = BOOLEAN_NORMALIZER_SETUP + "(begin "
    for t in terms:
        sexp = term_to_sexp(t)
        scheme += (f"(let-values (((result trace) (norm '{sexp})))"
                   f" (write result) (display #\\newline)) ")
    scheme += ")"

    output = run_wile(scheme)
    normalized_sexps = output.strip().split("\n")

    for i, term in enumerate(terms):
        original_sexp = term_to_sexp(term)
        original_table = truth_table(term)

        normalized_sexp = normalized_sexps[i]
        normalized_term = parse_sexp(normalized_sexp)
        normalized_table = truth_table(normalized_term)

        if original_table != normalized_table:
            failures += 1
            print(f"\n    FAIL: {original_sexp}")
            print(f"          normalized to {normalized_sexp}")
            print(f"          truth tables differ")
        elif args.snapshot and len(snapshot_cases) < 100:
            snapshot_cases.append((original_sexp, normalized_sexp))

    if failures == 0:
        print(f"OK ({num_terms} terms)")
    else:
        print(f"\n    {failures}/{num_terms} FAILED")

    if args.snapshot and snapshot_cases:
        body = BOOLEAN_NORMALIZER_SCHEME + "\n"
        for original, normalized in snapshot_cases:
            body += f";; {original} normalizes to {normalized}\n"
```
